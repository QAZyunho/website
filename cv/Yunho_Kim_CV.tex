% Title+date row as a two-column table (not \hfill) so a long title can
% wrap onto a second line without stranding the date off on its own.
% Body text is ragged-right throughout: fully-justified CJK/Latin mixed
% text can hit bad breakpoints (no valid stretch) and overflow the
% margin, so we trade a even right edge for text that never overflows.
% Titles additionally get hyphenpenalty=10000 locally (not document-wide)
% so a long title wraps at word boundaries instead of mid-word.
\newcommand{\entry}[3]{%
  \noindent\begin{tabularx}{\linewidth}{@{}>{\raggedright\arraybackslash}X r@{}}
    {\hyphenpenalty=10000\exhyphenpenalty=10000\textbf{#1}} & \textcolor{textgray}{#2}
  \end{tabularx}\\
  {\raggedright\textit{\textcolor{textgray}{#3}}\par}\vspace{0.5em}
}

\newcommand{\fund}[3]{%
  \noindent\begin{tabularx}{\linewidth}{@{}>{\raggedright\arraybackslash}X r@{}}
    {\hyphenpenalty=10000\exhyphenpenalty=10000\textbf{#1}} & \textcolor{textgray}{#2}
  \end{tabularx}\\
  {\raggedright\textcolor{textgray}{$\cdot$\ #3}\par}\vspace{0.5em}
}

\newcommand{\pub}[4]{%
  \textbf{#1}\\
  #2\\
  {\raggedright\textit{\textcolor{textgray}{#3}}#4\par}\vspace{0.6em}
}

% Korean-language entry: Malgun Gothic has no italic shape, so the venue
% line is set upright instead of silently falling back to regular weight
% mid-italic, which looked inconsistent next to the Latin entries above.
\newcommand{\pubkr}[4]{%
  {\cjkfont\bfseries #1}\\
  {\cjkfont #2}\\
  {\raggedright{\cjkfont\textcolor{textgray}{#3}}#4\par}\vspace{0.6em}
}
